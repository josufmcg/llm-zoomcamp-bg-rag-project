### Step 6: Create `src/bg_rag/config.py`

This module provides all configuration via environment variables using pydantic-settings:

```python
from pathlib import Path
from pydantic_settings import BaseSettings


class Settings(BaseSettings):
    """Application settings loaded from environment variables."""

    # PostgreSQL
    postgres_host: str = "localhost"
    postgres_port: int = 5432
    postgres_db: str = "bg_rag"
    postgres_user: str = "user"
    postgres_password: str = "password"

    # OpenAI
    openai_api_key: str = ""

    # Embedding model
    embedding_model_path: Path = Path("models/Xenova/all-MiniLM-L6-v2")

    # LLM
    llm_model: str = "gpt-4.1-mini"

    @property
    def database_url(self) -> str:
        """Build PostgreSQL connection string."""
        return (
            f"host={self.postgres_host} "
            f"port={self.postgres_port} "
            f"dbname={self.postgres_db} "
            f"user={self.postgres_user} "
            f"password={self.postgres_password}"
        )

    model_config = {
        "env_file": ".env",
        "env_file_encoding": "utf-8",
    }


def get_settings() -> Settings:
    """Create and return a Settings instance."""
    return Settings()
```

### Step 7: Create `scripts/download_model.py`

This script downloads the ONNX embedding model from HuggingFace Hub. It is adapted from the reference project's `02.vector/download.py`:

```python
"""Download the ONNX embedding model from HuggingFace Hub."""

import os
import shutil
import logging
from pathlib import Path
from huggingface_hub import hf_hub_download, list_repo_files

os.environ["HF_HUB_DISABLE_TELEMETRY"] = "1"
logging.getLogger("huggingface_hub").setLevel(logging.ERROR)

ONNX_CANDIDATES = [
    "onnx/model.onnx",
    "onnx/encoder_model.onnx",
    "model.onnx",
]


def download(repo: str, dest: str = "models") -> None:
    """Download ONNX model and tokenizer from a HuggingFace repo."""
    dest_path = Path(dest) / repo
    dest_path.mkdir(parents=True, exist_ok=True)

    files = list_repo_files(repo_id=repo)
    onnx_file = next((c for c in ONNX_CANDIDATES if c in files), None)
    if not onnx_file:
        raise FileNotFoundError(f"No ONNX model found in {repo}")

    for remote, local in [
        ("tokenizer.json", "tokenizer.json"),
        (onnx_file, "model.onnx"),
    ]:
        src = hf_hub_download(repo_id=repo, filename=remote)
        dst = dest_path / local
        if not dst.exists():
            shutil.copy2(src, dst)
            print(f"  saved {dst}")
        else:
            print(f"  exists {dst}")

    # Some models have an external data file
    onnx_ext = onnx_file + "_data"
    if onnx_ext in files:
        src = hf_hub_download(repo_id=repo, filename=onnx_ext)
        dst = dest_path / "model.onnx_data"
        if not dst.exists():
            shutil.copy2(src, dst)
            print(f"  saved {dst}")
        else:
            print(f"  exists {dst}")


if __name__ == "__main__":
    print("Downloading Xenova/all-MiniLM-L6-v2...")
    download("Xenova/all-MiniLM-L6-v2")
    print("Done.")
```

### Step 8: Create `Makefile`

Create the Makefile at the project root. This is a skeleton that will be expanded in later tasks:

```makefile
.PHONY: setup db init-db ingest dev-api dev-streamlit build up down logs ground-truth evaluate sample-data

# ── Setup ──────────────────────────────────────────
setup:
	uv sync
	uv run python scripts/download_model.py

# ── Local Development ──────────────────────────────
db:
	docker run -it --rm \
		--name bg-rag-postgres \
		-e POSTGRES_USER=user \
		-e POSTGRES_PASSWORD=password \
		-e POSTGRES_DB=bg_rag \
		-v bg_rag_pgdata:/var/lib/postgresql/data \
		-p 5432:5432 \
		pgvector/pgvector:pg17

init-db:
	uv run python -c "from bg_rag.db import init_db; init_db()"

ingest:
	uv run python -m bg_rag.ingest

dev-api:
	uv run uvicorn bg_rag.api.main:app --reload --host 0.0.0.0 --port 8000

dev-streamlit:
	cd src && uv run streamlit run bg_rag/frontend/app.py --server.port 8501

# ── Docker ─────────────────────────────────────────
build:
	docker compose build

up:
	docker compose up -d

down:
	docker compose down

logs:
	docker compose logs -f

# ── Evaluation ─────────────────────────────────────
ground-truth:
	uv run python scripts/generate_ground_truth.py

evaluate:
	uv run python scripts/evaluate.py

sample-data:
	uv run python scripts/generate_sample_data.py
```
